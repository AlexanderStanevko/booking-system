using Microsoft.AspNetCore.Diagnostics;

namespace MeetingRooms.Api.Common.Errors;

public sealed class GlobalExceptionHandler(ILogger<GlobalExceptionHandler> logger) : IExceptionHandler
{
    public async ValueTask<bool> TryHandleAsync(
        HttpContext context, Exception exception, CancellationToken cancellationToken)
    {
        var error = exception as AppException;
        var badRequest = exception as BadHttpRequestException;
        context.Response.StatusCode = error?.StatusCode
            ?? badRequest?.StatusCode ?? StatusCodes.Status500InternalServerError;

        if (error is null && badRequest is null)
        {
            logger.LogError(exception, "An unexpected API error occurred.");
        }

        var response = error is not null
            ? new ApiErrorResponse(error.Code, error.Message)
            : new ApiErrorResponse(
                badRequest is not null ? ErrorCodes.InvalidRequest : ErrorCodes.UnexpectedError,
                badRequest is not null ? "The request is invalid." : "An unexpected server error occurred.");

        await context.Response.WriteAsJsonAsync(response, cancellationToken);
        return true;
    }
}

using MeetingRooms.Api.Common.Errors;

namespace MeetingRooms.Api.Common.Time;

public static class TimeInterval
{
    public static bool Overlaps(
        DateTimeOffset startA, DateTimeOffset endA, DateTimeOffset startB, DateTimeOffset endB)
        => startA < endA && startB < endB && startA < endB && startB < endA;

    public static bool Intersects(
        DateTimeOffset start, DateTimeOffset end, DateTimeOffset? from, DateTimeOffset? to)
        => Overlaps(start, end, from ?? DateTimeOffset.MinValue, to ?? DateTimeOffset.MaxValue);

    public static double HoursWithin(
        DateTimeOffset start, DateTimeOffset end, DateTimeOffset? from, DateTimeOffset? to)
    {
        var clippedStart = from.HasValue && from > start ? from.Value : start;
        var clippedEnd = to.HasValue && to < end ? to.Value : end;
        return Math.Max(0, (clippedEnd - clippedStart).TotalHours);
    }

    public static void ValidateRange(DateTimeOffset? from, DateTimeOffset? to)
    {
        if (from.HasValue && to.HasValue && to <= from)
        {
            throw new AppException(ErrorCodes.InvalidTimeRange, "The range end must be later than its start.");
        }
    }
}

using System.Globalization;
using System.Text.RegularExpressions;

namespace MeetingRooms.Api.Common.Time;

public static class UtcTimestamp
{
    public static bool TryParse(string? value, out DateTimeOffset timestamp)
    {
        timestamp = default;
        if (value is null || !Regex.IsMatch(value, @"(?:[zZ]|[+-]\d{2}:\d{2})$"))
        {
            return false;
        }

        if (!DateTimeOffset.TryParse(value, CultureInfo.InvariantCulture, DateTimeStyles.None, out var parsed))
        {
            return false;
        }

        timestamp = parsed.ToUniversalTime();
        return true;
    }
}

using System.Globalization;
using System.Text.Json;
using System.Text.Json.Serialization;

namespace MeetingRooms.Api.Common.Time;

public sealed class UtcTimestampJsonConverter : JsonConverter<DateTimeOffset>
{
    public override DateTimeOffset Read(ref Utf8JsonReader reader, Type type, JsonSerializerOptions options)
    {
        if (reader.TokenType != JsonTokenType.String || !UtcTimestamp.TryParse(reader.GetString(), out var value))
        {
            throw new JsonException("Timestamps must include Z or an explicit UTC offset.");
        }

        return value;
    }

    public override void Write(Utf8JsonWriter writer, DateTimeOffset value, JsonSerializerOptions options)
        => writer.WriteStringValue(value.ToUniversalTime().ToString("O", CultureInfo.InvariantCulture));
}

using Microsoft.AspNetCore.Mvc.ModelBinding;

namespace MeetingRooms.Api.Common.Time;

public sealed class UtcTimestampModelBinder : IModelBinder
{
    public Task BindModelAsync(ModelBindingContext context)
    {
        var value = context.ValueProvider.GetValue(context.ModelName);
        if (value == ValueProviderResult.None)
        {
            return Task.CompletedTask;
        }

        context.ModelState.SetModelValue(context.ModelName, value);
        if (UtcTimestamp.TryParse(value.FirstValue, out var timestamp))
        {
            context.Result = ModelBindingResult.Success(timestamp);
        }
        else
        {
            context.ModelState.TryAddModelError(context.ModelName, "Timestamps must include Z or an explicit UTC offset.");
        }

        return Task.CompletedTask;
    }
}

    {
        services.AddControllers().AddJsonOptions(options =>
        {
            options.JsonSerializerOptions.Converters.Add(new JsonStringEnumConverter(allowIntegerValues: false));
            options.JsonSerializerOptions.Converters.Add(new UtcTimestampJsonConverter());
            options.JsonSerializerOptions.UnmappedMemberHandling = JsonUnmappedMemberHandling.Disallow;
        }).ConfigureApiBehaviorOptions(options =>
        {
            options.SuppressMapClientErrors = true;
            options.InvalidModelStateResponseFactory = _ => new BadRequestObjectResult(
                new ApiErrorResponse(ErrorCodes.InvalidRequest, "The request contains invalid or missing values."));
        });

        services.AddExceptionHandler<GlobalExceptionHandler>();
        services.AddProblemDetails();

            .ValidateDataAnnotations().ValidateOnStart();

        services.AddScoped<ReservationService>();

        services.AddScoped<ReservationValidator>();
        services.AddScoped<ReservationConflictChecker>();
        services.AddScoped<ReservationCompletion>();

        services.AddHostedService<ReservationCompletionWorker>();
        return services;
    }

using Microsoft.EntityFrameworkCore;
using MeetingRooms.Api.Models;

namespace MeetingRooms.Api.Data;

public sealed class ApplicationDbContext(DbContextOptions<ApplicationDbContext> options)
    : DbContext(options)
{
    public DbSet<Room> Rooms => Set<Room>();
    public DbSet<User> Users => Set<User>();
    public DbSet<Reservation> Reservations => Set<Reservation>();

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        modelBuilder.Entity<Room>().Property(room => room.Name).IsRequired().HasMaxLength(100);
        modelBuilder.Entity<Room>().Property(room => room.Location).IsRequired().HasMaxLength(200);
        modelBuilder.Entity<User>().Property(user => user.FirstName).IsRequired().HasMaxLength(100);
        modelBuilder.Entity<User>().Property(user => user.LastName).IsRequired().HasMaxLength(100);
        modelBuilder.Entity<Reservation>().Property(reservation => reservation.Title).IsRequired().HasMaxLength(200);
        modelBuilder.Entity<Reservation>().HasOne(reservation => reservation.Room)
            .WithMany(room => room.Reservations).HasForeignKey(reservation => reservation.RoomId)
            .OnDelete(DeleteBehavior.Restrict);
        modelBuilder.Entity<Reservation>().HasOne(reservation => reservation.Organizer)
            .WithMany(user => user.Reservations).HasForeignKey(reservation => reservation.OrganizerId)
            .OnDelete(DeleteBehavior.Restrict);
    }
}

using MeetingRooms.Api.Models;
using Microsoft.EntityFrameworkCore;

namespace MeetingRooms.Api.Data;

public sealed class DatabaseSeeder(ApplicationDbContext database, TimeProvider clock)
{
    public async Task SeedAsync(CancellationToken cancellationToken = default)
    {
        await database.Database.EnsureCreatedAsync(cancellationToken);
        if (await database.Rooms.AnyAsync(cancellationToken))
        {
            return;
        }

        var rooms = CreateRooms();
        var users = CreateUsers();
        database.Rooms.AddRange(rooms);
        database.Users.AddRange(users);
        database.Reservations.AddRange(CreateReservations(rooms, users, clock.GetUtcNow()));
        await database.SaveChangesAsync(cancellationToken);
    }

    private static List<Room> CreateRooms()
    {
        string[] names = ["Amber", "Birch", "Cedar", "Delta", "Elm", "Forest", "Grove", "Harbor", "Iris", "Juniper"];
        int[] capacities = [4, 6, 8, 12, 16, 20, 6, 10, 8, 14];
        return names.Select((name, index) => new Room
        {
            Id = StableId(1, index),
            Name = name,
            Capacity = capacities[index],
            Floor = index / 4 + 1,
            Location = $"Kaunas office, {(index % 2 == 0 ? "east" : "west")} wing",
            Equipment = Equipment.Whiteboard | (index % 2 == 0 ? Equipment.Projector : Equipment.TvScreen)
                | (index % 3 == 0 ? Equipment.VideoConferencing : Equipment.None),
            IsActive = index < 8
        }).ToList();
    }

    private static List<User> CreateUsers()
    {
        string[] firstNames = ["Alex", "Maya", "Daniel", "Emma", "Oliver", "Sofia", "Lukas", "Amelia",
            "Noah", "Isabella", "Leo", "Charlotte", "Ethan", "Ava", "James"];
        string[] lastNames = ["Morgan", "Patel", "Reed", "Wilson", "Bennett", "Martin", "Clark", "Walker",
            "Young", "Hall", "Allen", "King", "Wright", "Scott", "Green"];
        return firstNames.Select((name, index) => new User
        {
            Id = StableId(2, index),
            FirstName = name,
            LastName = lastNames[index]
        }).ToList();
    }

    private static List<Reservation> CreateReservations(List<Room> rooms, List<User> users, DateTimeOffset now)
    {
        string[] titles = ["Sprint planning", "Design review", "Team retrospective", "Project kickoff",
            "Customer workshop", "Architecture review", "Weekly team sync", "Interview panel"];
        var reservations = new List<Reservation>();
        for (var index = 0; index < 24; index++)
        {
            var upcoming = index < 8;
            var cancelled = index is >= 16 and < 20;
            var earlyCompleted = index >= 20;
            var room = rooms[index % 8];
            var user = users[index % users.Count];
            var start = index == 0 ? now.AddMinutes(-30)
                : upcoming || cancelled ? now.AddHours(1 + index * 2) : now.AddDays(-1 - index).AddHours(1);
            reservations.Add(new Reservation
            {
                Id = StableId(3, index),
                RoomId = room.Id,
                OrganizerId = user.Id,
                Room = room,
                Organizer = user,
                Title = titles[index % titles.Length],
                StartTime = start,
                EndTime = start.AddHours(2),
                AttendeeCount = Math.Min(room.Capacity, 3 + index % 6),
                Status = cancelled ? ReservationStatus.Cancelled
                    : upcoming ? ReservationStatus.Booked : ReservationStatus.Completed,
                ActualEndTime = earlyCompleted ? start.AddMinutes(45) : null
            });
        }

        return reservations;
    }

    private static Guid StableId(int category, int index)
        => Guid.Parse($"{category}0000000-0000-0000-0000-{index + 1:D12}");
}

    {

            .ToListAsync(cancellationToken);

using MeetingRooms.Api.Configuration;
using Microsoft.Extensions.Options;

namespace MeetingRooms.Api.Services.Reservations;

public sealed class ReservationCompletionWorker(
    IServiceScopeFactory scopeFactory, ReservationLock reservationLock, TimeProvider clock,
    IOptions<BookingOptions> options, ILogger<ReservationCompletionWorker> logger) : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        using var timer = new PeriodicTimer(TimeSpan.FromSeconds(options.Value.CompletionIntervalSeconds), clock);
        do
        {
            try
            {
                using var lease = await reservationLock.AcquireAsync(stoppingToken);
                await using var scope = scopeFactory.CreateAsyncScope();
                var completion = scope.ServiceProvider.GetRequiredService<ReservationCompletion>();
                await completion.FinalizeExpiredAsync(clock.GetUtcNow(), stoppingToken);
            }
            catch (OperationCanceledException) when (stoppingToken.IsCancellationRequested)
            {
                return;
            }
            catch (Exception exception)
            {
                logger.LogError(exception, "Failed to finalize expired reservations.");
            }
        }
        while (await timer.WaitForNextTickAsync(stoppingToken));
    }
}

using MeetingRooms.Api.Common.Errors;
using MeetingRooms.Api.Common.Time;
using MeetingRooms.Api.Data;
using MeetingRooms.Api.Models;
using Microsoft.EntityFrameworkCore;

namespace MeetingRooms.Api.Services.Reservations;

public sealed class ReservationConflictChecker(ApplicationDbContext database)
{
    public async Task ValidateAsync(Reservation candidate, CancellationToken cancellationToken = default)
    {
        var reservations = await database.Reservations.AsNoTracking()
            .Where(reservation => reservation.Id != candidate.Id
                && reservation.Status != ReservationStatus.Cancelled
                && (reservation.RoomId == candidate.RoomId || reservation.OrganizerId == candidate.OrganizerId))
            .ToListAsync(cancellationToken);

        var overlaps = reservations.Where(reservation => TimeInterval.Overlaps(
            candidate.StartTime, candidate.EndTime, reservation.StartTime,
            reservation.ActualEndTime ?? reservation.EndTime)).ToList();

        if (overlaps.Any(reservation => reservation.RoomId == candidate.RoomId))
        {
            throw new AppException(ErrorCodes.RoomAlreadyBooked,
                "This room is already booked during the selected time range.", StatusCodes.Status409Conflict);
        }

        if (overlaps.Any(reservation => reservation.OrganizerId == candidate.OrganizerId))
        {
            throw new AppException(ErrorCodes.OrganizerAlreadyBooked,
                "This organizer already has a booking during the selected time range.", StatusCodes.Status409Conflict);
        }
    }
}

namespace MeetingRooms.Api.Services.Reservations;

public sealed class ReservationLock : IDisposable
{
    private readonly SemaphoreSlim semaphore = new(1, 1);

    public async Task<IDisposable> AcquireAsync(CancellationToken cancellationToken = default)
    {
        await semaphore.WaitAsync(cancellationToken);
        return new Lease(semaphore);
    }

    public void Dispose() => semaphore.Dispose();

    private sealed class Lease(SemaphoreSlim semaphore) : IDisposable
    {
        private int released;

        public void Dispose()
        {
            if (Interlocked.Exchange(ref released, 1) == 0)
            {
                semaphore.Release();
            }
        }
    }
}

using MeetingRooms.Api.Common.Errors;
using MeetingRooms.Api.Data;
using MeetingRooms.Api.DTOs;
using MeetingRooms.Api.Mappings;
using MeetingRooms.Api.Models;
using Microsoft.EntityFrameworkCore;

namespace MeetingRooms.Api.Services.Reservations;

public sealed class ReservationService(
    ApplicationDbContext database, ReservationValidator validator, ReservationCompletion completion,
    ReservationLock reservationLock, TimeProvider clock)
{
    public async Task<ReservationResponse> CreateAsync(
        CreateReservationRequest request, CancellationToken cancellationToken = default)
    {
        using var lease = await reservationLock.AcquireAsync(cancellationToken);
        var now = clock.GetUtcNow();
        await completion.FinalizeExpiredAsync(now, cancellationToken);

        var reservation = request.ToEntity();
        (reservation.Room, reservation.Organizer) = await validator.ValidateAsync(reservation, now, cancellationToken);
        database.Reservations.Add(reservation);
        await database.SaveChangesAsync(cancellationToken);
        return reservation.ToResponse();
    }

    public async Task<ReservationResponse> UpdateAsync(
        Guid id, UpdateReservationRequest request, CancellationToken cancellationToken = default)
    {
        using var lease = await reservationLock.AcquireAsync(cancellationToken);
        var now = clock.GetUtcNow();
        await completion.FinalizeExpiredAsync(now, cancellationToken);
        var reservation = await FindAsync(id, cancellationToken);
        RequireUpcoming(reservation, now);

        var updated = request.ToEntity(id);
        var (room, organizer) = await validator.ValidateAsync(updated, now, cancellationToken);
        database.Entry(reservation).CurrentValues.SetValues(updated);
        reservation.Room = room;
        reservation.Organizer = organizer;
        await database.SaveChangesAsync(cancellationToken);
        return reservation.ToResponse();
    }

    public async Task<ReservationResponse> CancelAsync(Guid id, CancellationToken cancellationToken = default)
    {
        using var lease = await reservationLock.AcquireAsync(cancellationToken);
        var now = clock.GetUtcNow();
        await completion.FinalizeExpiredAsync(now, cancellationToken);
        var reservation = await FindAsync(id, cancellationToken);
        RequireUpcoming(reservation, now);

        reservation.Status = ReservationStatus.Cancelled;
        await database.SaveChangesAsync(cancellationToken);
        return reservation.ToResponse();
    }

    public async Task<ReservationResponse> CompleteAsync(Guid id, CancellationToken cancellationToken = default)
    {
        using var lease = await reservationLock.AcquireAsync(cancellationToken);
        var now = clock.GetUtcNow();
        await completion.FinalizeExpiredAsync(now, cancellationToken);
        var reservation = await FindAsync(id, cancellationToken);

        if (reservation.Status != ReservationStatus.Booked || reservation.StartTime > now || reservation.EndTime <= now)
        {
            throw new AppException(ErrorCodes.InvalidReservationStatus,
                "Only an ongoing booked meeting can be ended early.", StatusCodes.Status409Conflict);
        }

        reservation.Status = ReservationStatus.Completed;
        reservation.ActualEndTime = now;
        await database.SaveChangesAsync(cancellationToken);
        return reservation.ToResponse();
    }

    private async Task<Reservation> FindAsync(Guid id, CancellationToken cancellationToken)
        => await database.Reservations.Include(reservation => reservation.Room)
            .Include(reservation => reservation.Organizer)
            .SingleOrDefaultAsync(reservation => reservation.Id == id, cancellationToken)
            ?? throw new AppException(ErrorCodes.ReservationNotFound,
                "The requested reservation does not exist.", StatusCodes.Status404NotFound);

    private static void RequireUpcoming(Reservation reservation, DateTimeOffset now)
    {
        if (reservation.Status != ReservationStatus.Booked || reservation.StartTime <= now)
        {
            throw new AppException(ErrorCodes.InvalidReservationStatus,
                "Only an upcoming booked reservation can be edited or cancelled.", StatusCodes.Status409Conflict);
        }
    }
}

using MeetingRooms.Api.Common.Errors;
using MeetingRooms.Api.Configuration;
using MeetingRooms.Api.Data;
using MeetingRooms.Api.Models;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Options;

namespace MeetingRooms.Api.Services.Reservations;

public sealed class ReservationValidator(
    ApplicationDbContext database, IOptions<BookingOptions> options, ReservationConflictChecker conflicts)
{
    public async Task<(Room Room, User Organizer)> ValidateAsync(
        Reservation reservation, DateTimeOffset now, CancellationToken cancellationToken = default)
    {
        ValidateTimeRange(reservation, now);
        ValidateDuration(reservation);
        ValidateHorizon(reservation.StartTime, now);
        ValidateTitle(reservation.Title);

        var room = await database.Rooms.FindAsync([reservation.RoomId], cancellationToken)
            ?? throw new AppException(ErrorCodes.RoomNotFound, "The requested room does not exist.", StatusCodes.Status404NotFound);
        var organizer = await database.Users.FindAsync([reservation.OrganizerId], cancellationToken)
            ?? throw new AppException(ErrorCodes.UserNotFound, "The requested organizer does not exist.", StatusCodes.Status404NotFound);

        ValidateRoom(room);
        ValidateCapacity(reservation.AttendeeCount, room.Capacity);
        await conflicts.ValidateAsync(reservation, cancellationToken);
        return (room, organizer);
    }

    private static void ValidateTimeRange(Reservation reservation, DateTimeOffset now)
    {
        if (reservation.EndTime <= reservation.StartTime)
        {
            throw new AppException(ErrorCodes.InvalidTimeRange, "The end time must be later than the start time.");
        }

        if (reservation.StartTime <= now)
        {
            throw new AppException(ErrorCodes.StartTimeNotInFuture, "The reservation must start in the future.");
        }
    }

    private void ValidateDuration(Reservation reservation)
    {
        if ((reservation.EndTime - reservation.StartTime).TotalDays > options.Value.MaxDurationDays)
        {
            throw new AppException(ErrorCodes.MaxDurationExceeded,
                $"A reservation cannot last longer than {options.Value.MaxDurationDays} days.");
        }
    }

    private void ValidateHorizon(DateTimeOffset start, DateTimeOffset now)
    {
        if ((start - now).TotalDays > options.Value.MaxAdvanceBookingDays)
        {
            throw new AppException(ErrorCodes.BookingHorizonExceeded,
                $"A reservation cannot start more than {options.Value.MaxAdvanceBookingDays} days in the future.");
        }
    }

    private static void ValidateCapacity(int attendees, int capacity)
    {
        if (attendees <= 0)
        {
            throw new AppException(ErrorCodes.InvalidAttendeeCount, "The attendee count must be greater than zero.");
        }

        if (attendees > capacity)
        {
            throw new AppException(ErrorCodes.RoomCapacityExceeded, "The attendee count exceeds the room capacity.");
        }
    }

    private static void ValidateRoom(Room room)
    {
        if (!room.IsActive)
        {
            throw new AppException(ErrorCodes.RoomInactive, "This room is inactive and cannot be booked.");
        }
    }

    private static void ValidateTitle(string title)
    {
        if (string.IsNullOrWhiteSpace(title) || title.Length > 200)
        {
            throw new AppException(ErrorCodes.InvalidRequest, "The title must contain between 1 and 200 characters.");
        }
    }
}

using MeetingRooms.Api.Common.Time;
using MeetingRooms.Api.Data;
using MeetingRooms.Api.DTOs;
using MeetingRooms.Api.Models;
using MeetingRooms.Api.Services.Reservations;
using Microsoft.EntityFrameworkCore;

namespace MeetingRooms.Api.Services;

public sealed class RoomUsageService(
    ApplicationDbContext database, ReservationCompletion completion, ReservationLock reservationLock, TimeProvider clock)
{
    public async Task<IReadOnlyList<RoomUsageResponse>> GetAsync(
        DateRangeFilter filter, CancellationToken cancellationToken = default)
    {
        TimeInterval.ValidateRange(filter.From, filter.To);
        using var lease = await reservationLock.AcquireAsync(cancellationToken);
        await completion.FinalizeExpiredAsync(clock.GetUtcNow(), cancellationToken);
        var rooms = await database.Rooms.AsNoTracking().OrderBy(room => room.Name).ToListAsync(cancellationToken);
        var reservations = await database.Reservations.AsNoTracking().ToListAsync(cancellationToken);
        var byRoom = reservations.Where(reservation => TimeInterval.Intersects(
            reservation.StartTime, reservation.EndTime, filter.From, filter.To)).ToLookup(reservation => reservation.RoomId);
        return rooms.Select(room => Calculate(room, byRoom[room.Id].ToList(), filter)).ToList();
    }

    private static RoomUsageResponse Calculate(Room room, List<Reservation> reservations, DateRangeFilter filter)
    {
        var bookedHours = reservations.Where(reservation => reservation.Status != ReservationStatus.Cancelled)
            .Sum(reservation => TimeInterval.HoursWithin(reservation.StartTime, reservation.EndTime, filter.From, filter.To));
        var completed = reservations.Where(reservation => reservation.Status == ReservationStatus.Completed).ToList();
        var confirmedHours = completed.Where(reservation => reservation.ActualEndTime.HasValue)
            .Sum(reservation => TimeInterval.HoursWithin(reservation.StartTime,
                reservation.ActualEndTime!.Value, filter.From, filter.To));
        var estimatedHours = completed.Where(reservation => !reservation.ActualEndTime.HasValue)
            .Sum(reservation => TimeInterval.HoursWithin(reservation.StartTime, reservation.EndTime, filter.From, filter.To));

        return new RoomUsageResponse(room.Id, room.Name, reservations.Count,
            reservations.Count(reservation => reservation.Status == ReservationStatus.Cancelled), completed.Count,
            bookedHours, confirmedHours, estimatedHours, bookedHours > 0 ? confirmedHours / bookedHours * 100 : 0);
    }
}

    private sealed class ObservedClock : TimeProvider

        public TaskCompletionSource TimerCreated { get; } = new(TaskCreationOptions.RunContinuationsAsynchronously);
        public TaskCompletionSource FirstRead { get; } = new(TaskCreationOptions.RunContinuationsAsynchronously);
        public bool ObserveReads { get; set; }

        public override DateTimeOffset GetUtcNow()
        {
            var now = Inner.GetUtcNow();
            if (ObserveReads)

                FirstRead.TrySetResult();

            }

namespace MeetingRooms.Api.Tests;

    public async Task Application_errors_preserve_their_code_message_and_http_status()
    {
        var exception = new AppException(ErrorCodes.RoomAlreadyBooked, "The room is already booked.", StatusCodes.Status409Conflict);
        var (status, error) = await HandleAsync(exception);
        Assert.Equal(StatusCodes.Status409Conflict, status);
        Assert.Equal(exception.Code, error.Code);
        Assert.Equal(exception.Message, error.Message);
    }

    [Fact]
    public async Task Unexpected_errors_return_a_safe_structured_server_error()
    {

        Assert.Equal(StatusCodes.Status500InternalServerError, status);
        Assert.Equal(ErrorCodes.UnexpectedError, error.Code);

    }

    private static async Task<(int Status, ApiErrorResponse Error)> HandleAsync(Exception exception)
    {
        await using var services = new ServiceCollection().AddOptions().BuildServiceProvider();
        var context = new DefaultHttpContext { RequestServices = services };
        await using var body = new MemoryStream();
        context.Response.Body = body;
        var handler = new GlobalExceptionHandler(NullLogger<GlobalExceptionHandler>.Instance);
        Assert.True(await handler.TryHandleAsync(context, exception, CancellationToken.None));
        body.Position = 0;
        var error = await JsonSerializer.DeserializeAsync<ApiErrorResponse>(body, new JsonSerializerOptions(JsonSerializerDefaults.Web));
        return (context.Response.StatusCode, error!);
    }

using System.Text.Json;
using MeetingRooms.Api.Common.Time;

namespace MeetingRooms.Api.Tests;

public sealed class TimestampTests
{
    [Theory]
    [InlineData("2030-01-10T12:00:00Z")]
    [InlineData("2030-01-10T15:00:00+03:00")]
    [InlineData("2030-01-10T07:00:00-05:00")]
    public void Timestamp_parsing_preserves_the_instant_and_normalizes_to_utc(string text)
    {
        Assert.True(UtcTimestamp.TryParse(text, out var timestamp));
        Assert.Equal(new DateTimeOffset(2030, 1, 10, 12, 0, 0, TimeSpan.Zero), timestamp);
        Assert.Equal(TimeSpan.Zero, timestamp.Offset);
    }

    [Theory]
    [InlineData("2030-01-10T12:00:00")]
    [InlineData("2030-01-10")]
    [InlineData("invalid")]
    [InlineData("2030-01-10T12:00:00+25:00")]
    public void Ambiguous_or_invalid_timestamps_are_rejected(string text)
    {
        Assert.False(UtcTimestamp.TryParse(text, out _));
        var options = new JsonSerializerOptions();
        options.Converters.Add(new UtcTimestampJsonConverter());
        Assert.Throws<JsonException>(() => JsonSerializer.Deserialize<DateTimeOffset>(JsonSerializer.Serialize(text), options));
    }

    [Fact]
    public void Zero_length_effective_intervals_never_block_other_bookings()
    {
        var start = new DateTimeOffset(2030, 1, 10, 12, 0, 0, TimeSpan.Zero);
        Assert.False(TimeInterval.Overlaps(start, start, start.AddHours(-1), start.AddHours(1)));
    }
